── REGISTERS ──────────────────────────────────────────────────

  32-bit    16-bit    8-bit (high/low)    Purpose
  ─────────────────────────────────────────────────────
  eax       ax        ah/al               Accumulator
  ebx       bx        bh/bl               Base
  ecx       cx        ch/cl               Counter
  edx       dx        dh/dl               Data
  esi       si        sil                 Source Index
  edi       di        dil                 Dest Index
  ebp       bp        bpl                 Base Pointer
  esp       sp        spl                 Stack Pointer

  Segment: cs ds es fs gs ss
  EIP = instruction pointer, EFLAGS = flags

── CALLING CONVENTIONS ────────────────────────────────────────

  cdecl (C default, 32-bit):
    Args pushed right-to-left on stack
    Caller cleans stack
    Return: eax (edx:eax for 64-bit)
    Caller-saved: eax, ecx, edx
    Callee-saved: ebx, esi, edi, ebp

  stdcall (Win32 API):
    Args pushed right-to-left on stack
    Callee cleans stack (ret N)
    Return: eax
    Same register rules as cdecl

  fastcall (MS):
    First 2 args in ecx, edx → rest on stack
    Callee cleans stack
    Return: eax

  thiscall (MS C++):
    this in ecx, args on stack
    Callee cleans stack

  System calls (Linux int 0x80):
    eax = syscall number
    Args: ebx, ecx, edx, esi, edi, ebp
    Return: eax

── DATA SIZES ─────────────────────────────────────────────────

  byte    = 8 bits    (db)
  word    = 16 bits   (dw)
  dword   = 32 bits   (dd)
  qword   = 64 bits   (dq, FPU only)
  tword   = 80 bits   (dt, FPU only)

── ADDRESSING MODES ──────────────────────────────────────────

  mov eax, [ebx]              ; base
  mov eax, [ebx + 4]          ; base + displacement
  mov eax, [ebx + ecx*4]      ; base + index*scale
  mov eax, [ebx + ecx*4 + 8]  ; base + index*scale + disp
  mov eax, [label]            ; absolute address

  Scale: 1, 2, 4, or 8
  Any 32-bit GPR can be base or index (except ESP as index)

── DATA MOVEMENT ──────────────────────────────────────────────

  mov   dst, src              ; move
  movzx dst, src              ; zero-extend
  movsx dst, src              ; sign-extend
  lea   dst, [mem]            ; load effective address
  xchg  dst, src              ; swap
  cmovcc dst, src             ; conditional move (Pentium Pro+)
  push  src                   ; push dword
  pop   dst                   ; pop dword
  pushad                      ; push all 8 GPRs (28h bytes)
  popad                       ; pop all 8 GPRs
  pushfd                      ; push EFLAGS
  popfd                       ; pop EFLAGS
  lds/les/lfs/lgs/lss         ; load far pointer (seg:offset)

── ARITHMETIC ─────────────────────────────────────────────────

  add   dst, src              ; dst += src
  adc   dst, src              ; dst += src + CF
  sub   dst, src              ; dst -= src
  sbb   dst, src              ; dst -= src + CF
  inc   dst                   ; dst++
  dec   dst                   ; dst--
  neg   dst                   ; negate
  cmp   op1, op2              ; compare (sets flags)

  mul   src                   ; unsigned: edx:eax = eax * src
  imul  src                   ; signed:   edx:eax = eax * src
  imul  dst, src              ; dst = dst * src
  imul  dst, src, imm         ; dst = src * imm

  div   src                   ; unsigned: eax=quot, edx=rem (of edx:eax)
  idiv  src                   ; signed version
  cdq                         ; sign-extend eax into edx:eax (before div)

  aaa/aas/aam/aad             ; ASCII adjust (BCD arithmetic)
  daa/das                     ; decimal adjust (packed BCD)

── LOGICAL / BITWISE ──────────────────────────────────────────

  and   dst, src
  or    dst, src
  xor   dst, src              ; xor eax, eax = zero (preferred)
  not   dst
  test  op1, op2              ; AND w/o store (sets flags)

  shl   dst, count            ; shift left
  shr   dst, count            ; shift right (unsigned)
  sar   dst, count            ; shift right (signed, arithmetic)
  sal   dst, count            ; same as shl
  rol   dst, count            ; rotate left
  ror   dst, count            ; rotate right
  rcl   dst, count            ; rotate through carry left
  rcr   dst, count            ; rotate through carry right
  shld  dst, src, count       ; double-precision shift left
  shrd  dst, src, count       ; double-precision shift right

  bsf   dst, src              ; bit scan forward
  bsr   dst, src              ; bit scan reverse
  bt    bit, base             ; bit test
  bts   bit, base             ; bit test and set
  btr   bit, base             ; bit test and reset
  btc   bit, base             ; bit test and complement

── FLAGS (EFLAGS) ─────────────────────────────────────────────

── CONDITIONAL JUMPS ──────────────────────────────────────────

  Mnemonic        Condition               After CMP A, B
  ─────────────────────────────────────────────────────
  je / jz         ZF=1                    A == B
  jne / jnz       ZF=0                    A != B
  jg / jnle       ZF=0 && SF=OF           A > B  (signed)
  jge / jnl       SF=OF                   A >= B (signed)
  ja / jnbe       CF=0 && ZF=0           A > B  (unsigned)
  jae / jnb       CF=0                    A >= B (unsigned)
  jl / jnge       SF!=OF                  A < B  (signed)
  jle / jng       ZF=1 || SF!=OF          A <= B (signed)
  jb / jnae       CF=1                    A < B  (unsigned)
  jbe / jna       CF=1 || ZF=1            A <= B (unsigned)
  jo              OF=1                    overflow
  jno             OF=0                    no overflow
  js              SF=1                    negative
  jns             SF=0                    non-negative
  jp / jpe        PF=1                    parity even
  jnp / jpo       PF=0                    parity odd
  jcxz            CX=0                    (16-bit)
  jecxz           ECX=0                   (32-bit)

  SETcc reg       ; set byte if condition true
  CMOVcc dst, src ; conditional move (Pentium Pro+)

── CONTROL FLOW ───────────────────────────────────────────────

  jmp   target                ; unconditional
  jmp   reg                   ; indirect jump (switch tables)
  call  target                ; push EIP, jump
  call  reg                   ; indirect call
  ret                         ; pop EIP
  ret   N                     ; pop EIP, esp += N (stdcall)
  int   0x80                  ; Linux syscall
  int   3                     ; debug breakpoint
  int   1                     ; single-step debug
  into                        ; interrupt on overflow
  iret                        ; return from interrupt
  loop  target                ; dec ecx, jnz target
  loope/loopz                 ; loop while equal/zero
  loopne/loopnz               ; loop while not equal/not zero
  enter N, 0                  ; stack frame setup
  leave                       ; mov esp, ebp; pop ebp

── COMMON PATTERNS ────────────────────────────────────────────

  ; Function prologue (cdecl)
      push ebp
      mov  ebp, esp
      sub  esp, N          ; N bytes of locals

  ; Function epilogue
      mov  esp, ebp
      pop  ebp
      ret

  ; Function epilogue (stdcall, callee cleans args)
      mov  esp, ebp
      pop  ebp
      ret  N              ; N = arg bytes to pop

  ; Access stack args (cdecl, after prologue)
      mov  eax, [ebp+8]    ; arg1
      mov  eax, [ebp+12]   ; arg2
      mov  eax, [ebp+16]   ; arg3

  ; Local vars
      mov  eax, [ebp-4]    ; local 1
      mov  eax, [ebp-8]    ; local 2

  ; Loop
      xor  ecx, ecx        ; i = 0
  .loop:
      ; ... body ...
      inc  ecx
      cmp  ecx, N
      jl   .loop

  ; While loop
  .top:
      test eax, eax
      jz   .done
      ; ... body ...
      jmp  .top
  .done:

  ; If / else
      cmp  eax, ebx
      jne  .else
      ; ... if body ...
      jmp  .endif
  .else:
      ; ... else body ...
  .endif:

  ; Switch (jump table)
      cmp  eax, MAX_CASE
      ja   .default
      jmp  [jump_table + eax*4]
  .case0: ...
  .case1: ...

  ; String length
      mov  edi, str
      xor  ecx, ecx
      not  ecx             ; ecx = -1 (max count)
      xor  al, al          ; search for null
      repne scasb
      not  ecx
      dec  ecx             ; length in ecx

  ; Memory copy
      mov  esi, src
      mov  edi, dst
      mov  ecx, len
      rep  movsb           ; copy ecx bytes

  ; Struct member access
      ; struct at [ebp+8], member offset 12
      mov  eax, [ebp+8]    ; struct ptr
      mov  ebx, [eax+12]   ; member

  ; Min / Max
      cmp  eax, ebx
      jle  .no_swap
      xchg eax, ebx
  .no_swap:

  ; Absolute value
      cdq                  ; edx = sign mask of eax
      xor  eax, edx
      sub  eax, edx

  ; Sign extend
      movsx eax, byte [val]  ; byte → dword
      movsx eax, word [val]  ; word → dword

  ; Linux syscall (int 0x80)
      mov  eax, 4           ; sys_write
      mov  ebx, 1           ; fd = stdout
      mov  ecx, msg         ; buffer
      mov  edx, len         ; length
      int  0x80

  ; Linux exit
      mov  eax, 1           ; sys_exit
      xor  ebx, ebx         ; status 0
      int  0x80

── SEGMENTS & SELECTORS ───────────────────────────────────────

  Real mode:  physical = segment * 16 + offset
  Protected:  segment → descriptor table → base + limit
  Flat model: cs/ds/es/fs/gs all base=0, limit=4GB

  GDT entry: base(32), limit(20), access(8), flags(4)
  LDT: local descriptor table per task
  TSS: task state segment

  Descriptor privilege: DPL 0 (kernel) .. 3 (user)
  CPL = current privilege level (low 2 bits of CS)
  RPL = requestor privilege level (low 2 bits of selector)

── PROTECTED MODE CONCEPTS ───────────────────────────────────

  CR0: PE (protected mode), PG (paging), WP (write protect)
  CR3: page directory physical address (PDBR)
  CR4: PAE, PSE, OSFXSR, OSXMMEXCPT

  Paging (4KB pages):
    32-bit linear → [PDE 10 bits][PTE 10 bits][offset 12 bits]
    PDE → page table base, PTE → physical page base
    PDE/PTE flags: P, R/W, U/S, PWT, PCD, A, D, PS

  PAE (36-bit physical):
    3-level: [PDPTE 2][PDE 9][PTE 9][offset 12]
    4KB pages, 64-bit PDE/PTE with NX bit

  PSE (4MB pages):
    PDE.PS=1 → single 4MB page, no PTE
